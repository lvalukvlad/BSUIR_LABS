# Класс нейронной сети
class NeuralNetwork(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim):
        """
        Инициализация слоёв нейронной сети

        Args:
            input_dim: количество признаков (входных нейронов)
            hidden_dim: количество нейронов в скрытом слое
            output_dim: количество классов (выходных нейронов)
        """
        super(NeuralNetwork, self).__init__()

        # Входной слой -> Скрытый слой
        self.fc1 = nn.Linear(input_dim, hidden_dim)

        # Функция активации ReLU (после скрытого слоя)
        self.relu = nn.ReLU()

        # Скрытый слой -> Выходной слой
        self.fc2 = nn.Linear(hidden_dim, output_dim)

    def forward(self, x):
        """
        Прямой проход (forward pass)

        Args:
            x: входные данные (тензор)

        Returns:
            output: выходные значения сети (логиты)
        """
        # Проход через первый полносвязный слой + активация
        x = self.fc1(x)
        x = self.relu(x)

        # Проход через выходной слой (без активации - для CrossEntropyLoss)
        x = self.fc2(x)
        return x

# Создание экземпляра модели

class NeuralNetwork_Activation(nn.Module):
    """
    Нейронная сеть с настраиваемой функцией активации
    """

    def __init__(self, input_dim, hidden_dim, output_dim, activation='relu'):
        super(NeuralNetwork_Activation, self).__init__()

        self.fc1 = nn.Linear(input_dim, hidden_dim)

        # Выбор функции активации
        if activation == 'relu':
            self.activation = nn.ReLU()
        elif activation == 'sigmoid':
            self.activation = nn.Sigmoid()
        elif activation == 'tanh':
            self.activation = nn.Tanh()
        elif activation == 'leaky_relu':
            self.activation = nn.LeakyReLU(negative_slope=0.01)
        else:
            raise ValueError(f"Неизвестная функция активации: {activation}")

        self.fc2 = nn.Linear(hidden_dim, output_dim)

    def forward(self, x):
        x = self.fc1(x)
        x = self.activation(x)
        x = self.fc2(x)
        return x

train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
test_dataset = TensorDataset(X_test_tensor, y_test_tensor)

train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
test_loader = DataLoader(test_dataset, batch_size=batch_size, shuffle=False)

print(f"\nDataLoader создан:")
print(f"   Train batches: {len(train_loader)}")
print(f"   Test batches: {len(test_loader)}")

# Инициализация истории обучения
# Списки для сохранения метрик по эпохам
train_loss_history = []
train_acc_history = []
test_loss_history = []
test_acc_history = []

# Отслеживание метрик
# Функция для вычисления accuracy
def calculate_accuracy(outputs, labels):
    _, predicted = torch.max(outputs, dim=1)
    accuracy = (predicted == labels).sum().item() / labels.size(0)
    return accuracy

print("\nСписки для метрик инициализированы")

# Цикл обучения (на 50 эпох)
print(f"\nЗапуск цикла обучения ({num_epochs} эпох)")

for epoch in range(num_epochs):

    # Обучение на train-выборке
    model.train()  # Режим обучения (включает dropout, batchnorm)

    epoch_train_loss = 0.0
    epoch_train_correct = 0
    epoch_train_total = 0

    for batch_X, batch_y in train_loader:
        # Шаг 1: Обнулить градиенты
        optimizer.zero_grad()

        # Шаг 2: Прямой проход (forward pass)
        outputs = model(batch_X)

        # Шаг 3: Вычислить функцию потерь
        loss = criterion(outputs, batch_y)

        # Шаг 4: Обратное распространение ошибки (backward pass)
        loss.backward()

        # Шаг 5: Обновить веса модели
        optimizer.step()

        # Накопление метрик для эпохи
        epoch_train_loss += loss.item() * batch_X.size(0)
        epoch_train_correct += calculate_accuracy(outputs, batch_y) * batch_y.size(0)
        epoch_train_total += batch_y.size(0)

    # Средние метрики по train выборке за эпоху
    avg_train_loss = epoch_train_loss / epoch_train_total
    avg_train_acc = epoch_train_correct / epoch_train_total

    # Оценка на test-выборке
    model.eval()  # Режим оценки (выключает dropout, batchnorm)

    epoch_test_loss = 0.0
    epoch_test_correct = 0
    epoch_test_total = 0

    with torch.no_grad():
        for batch_X, batch_y in test_loader:
            # Прямой проход
            outputs = model(batch_X)

            # Вычислить loss
            loss = criterion(outputs, batch_y)

            # Накопление метрик
            epoch_test_loss += loss.item() * batch_X.size(0)
            epoch_test_correct += calculate_accuracy(outputs, batch_y) * batch_y.size(0)
            epoch_test_total += batch_y.size(0)

    # Средние метрики по test выборке за эпоху
    avg_test_loss = epoch_test_loss / epoch_test_total
    avg_test_acc = epoch_test_correct / epoch_test_total

    # Сохранение истории
    train_loss_history.append(avg_train_loss)
    train_acc_history.append(avg_train_acc)
    test_loss_history.append(avg_test_loss)
    test_acc_history.append(avg_test_acc)

    # Вывод прогресса (каждые 10 эпох)
    if (epoch + 1) % 10 == 0 or epoch == 0:
        print(f"Эпоха {epoch + 1:3d}/{num_epochs} | "
              f"Train Loss: {avg_train_loss:.4f} | "
              f"Train Acc: {avg_train_acc:.4f} | "
              f"Test Loss: {avg_test_loss:.4f} | "

Почему CrossEntropyLoss?
   Подходит для многоклассовой классификации
   Включает в себя softmax (не нужно применять отдельно)
   Ожидает на входе:

   Вычисляет разницу между предсказанием и истинным классом

Почему результаты могут различаться:

1.  СЛОЖНОСТЬ МОДЕЛИ:
   KNeighbors: простая модель, основана на расстоянии между объектами
   Neural Network: способна выявлять сложные нелинейные зависимости
   Наша сеть: 1 скрытый слой (64 нейрона) — умеренная сложность

2.  КОЛИЧЕСТВО ДАННЫХ:
   Dataset: 100,000 объектов — достаточно для нейросети
   При меньшем объёме данных классические методы часто выигрывают
   Нейросети раскрывают потенциал на больших датасетах

3.  НЕЛИНЕЙНОСТЬ ПРИЗНАКОВ:
   ReLU активация позволяет моделировать нелинейные зависимости
   KNeighbors неявно учитывает нелинейность через метрику расстояния
   Логистическая регрессия — линейная модель (поэтому 1.00 — это аномалия)

4.  ПЕРЕОБУЧЕНИЕ:
   DecisionTree/LogReg с Accuracy=1.00 — явное переобучение или утечка данных
   Возможные причины утечки:
     - Признаки, напрямую связанные с целевой переменной
     - Неправильное разделение train/test (дубликаты в обеих выборках)
     - Масштабирование до разделения (data leakage)
   Наша нейросеть: более честная оценка (train/test разделены корректно)

5.  ГИПЕРПАРАМЕТРЫ:
   KNeighbors: n_neighbors=5 (может требовать подбора)
   Neural Network: 64 нейрона, lr=0.001, 50 эпох (можно улучшить)
   Дальнейшая настройка может улучшить результат нейросети
