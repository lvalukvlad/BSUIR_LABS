\sectioncentered*{Перечень условных обозначений}
\addcontentsline{toc}{section}{Перечень условных обозначений}
\label{sec:reduction}
\begin{flushleft}
	\setlength{\parindent}{1.25cm}
		\indent
	\indent API \mdash Application Programming Interface, программный интерфейс приложения;\\
	\indent REST \mdash Representational State Transfer, архитектурный стиль взаимодействия компонентов;\\
	\indent JSON \mdash JavaScript Object Notation, текстовый формат обмена данными;\\
	\indent HTTP \mdash HyperText Transfer Protocol, протокол передачи гипертекста;\\
	\indent SQL \mdash Structured Query Language, язык структурированных запросов;\\
	\indent СУБД \mdash система управления базами данных;\\
	\indent БД \mdash база данных;\\
	\indent ЛВС \mdash локальная вычислительная сеть;\\
	\indent ИПС \mdash информационно-поисковая система;\\
	\indent СИП \mdash система информационного поиска;\\
	\indent ЕЯ \mdash естественный язык;\\
	\indent ПОД \mdash поисковый образ документа;\\
	\indent ПОЗ \mdash поисковый образ запроса;\\
	\indent TF \mdash Term Frequency, частота термина в документе;\\
	\indent IDF \mdash Inverse Document Frequency, инверсная документная частота;\\
	\indent TF-IDF \mdash Term Frequency --- Inverse Document Frequency, статистическая мера значимости термина;\\
	\indent BM25 \mdash Best Match 25, вероятностная модель ранжирования Okapi;\\
	\indent BIM \mdash Binary Independence Model, бинарная модель независимости терминов;\\
	\indent RSJ \mdash Robertson--Sparck Jones, вес Робертсона -- Спарк Джонс;\\
	\indent RSV \mdash Retrieval Status Value, статус релевантности документа;\\
	\indent PRF \mdash Pseudo-Relevance Feedback, псевдорелевантная обратная связь;\\
	\indent MAP \mdash Mean Average Precision, средняя средняя точность;\\
	\indent nDCG \mdash Normalized Discounted Cumulative Gain, нормализованный дисконтированный накопленный выигрыш;\\
	\indent P@k \mdash Precision at k, точность на первых k результатах;\\
	\indent F-мера \mdash мера качества поиска, объединяющая точность и полноту;\\
	\indent NLP \mdash Natural Language Processing, обработка естественного языка;\\
	\indent SPA \mdash Single Page Application, одностраничное веб-приложение;\\
	\indent CSV \mdash Comma-Separated Values, формат с разделителями-запятыми;\\
	\indent PDF \mdash Portable Document Format, переносимый формат документов;\\
	\indent DOCX \mdash формат документов Microsoft Word;\\
	\indent RTF \mdash Rich Text Format, формат расширенного текста;\\
	\indent HTML \mdash HyperText Markup Language, язык гипертекстовой разметки.
\end{flushleft}
